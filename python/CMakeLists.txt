set(Python3_FIND_VIRTUALENV FIRST)
set(Python3_FIND_STRATEGY LOCATION)
set(PYBIND11_FINDPYTHON ON)
find_package(Python3 3.11 COMPONENTS Interpreter Development.Module)
option(PYTHON_ENFORCE_TESTS "Enforce Python tests" OFF)
if(Python3_FOUND)
    report_python_configuration_once()
    foreach(_required pybind11::module pyariadne-python-common pyariadne-evolution pyariadne-smt pyariadne-sat)
        if(NOT TARGET ${_required})
            message(FATAL_ERROR "Missing Python dependency: ${_required}")
        endif()
    endforeach()
    add_subdirectory(bindings)
    if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
        ariadne_finalize_python_module(NAME reasoning)
        if(WIN32)
            set(_python_install_dir "Lib/site-packages")
        else()
            set(_python_install_dir "lib/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages")
        endif()
        set(PYTHON_BINDINGS_INSTALL_DIR "${_python_install_dir}" CACHE STRING "Python extension install directory relative to prefix")
        install(TARGETS pyariadne DESTINATION "${PYTHON_BINDINGS_INSTALL_DIR}")
        add_subdirectory(tests)
        add_dependencies(bindings pyariadne)
    endif()
elseif(PYTHON_ENFORCE_TESTS)
    message(FATAL_ERROR "Python >= 3.11 with development files is required.")
endif()
